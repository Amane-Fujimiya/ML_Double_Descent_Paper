%################################################################################
% -----------------------------------------------
% PLAIN: asserted results — italic body text
% -----------------------------------------------
\theoremstyle{plain}
\newtheorem{theorem}{Theorem}[section]
\newtheorem{proposition}[theorem]{Proposition}
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{col}{Corollary}[section]
\newtheorem{claim}{Claim}[theorem]
\newtheorem{conjecture}{Conjecture}[section]
\newtheorem{hypothesis}{Hypothesis}[section]
\newtheorem{assumption}[theorem]{Assumption}
\newtheorem{assume}{Assumption}[subsection]

% -----------------------------------------------
% DEFINITION: constitutive content — upright body text
% -----------------------------------------------
\theoremstyle{definition}
\newtheorem{definition}{Definition}[section]
\newtheorem{axiom}{Axiom}[section]
\newtheorem{setting}{Setting}[section]
\newtheorem{construction}{Construction}[subsection]
\newtheorem{example}{Example}[section]

% -----------------------------------------------
% REMARK: peripheral commentary — italic label, upright body
% -----------------------------------------------
\theoremstyle{remark}
\newtheorem{remark}[theorem]{Remark}
\newtheorem{note}{Note}[section]
\newtheorem{question}{Question}[section]
